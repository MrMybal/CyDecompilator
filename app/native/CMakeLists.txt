cmake_minimum_required(VERSION 3.24)
project(CyDecompilatorMCP LANGUAGES CXX)
if(MSVC)
  # Exception unwinding is required by the SDK's cancellation/RAII contract,
  # including when an earlier incomplete configure left empty CMake flag caches.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/EHsc /utf-8 /Brepro /experimental:deterministic $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/DNDEBUG>)
  add_compile_options(/pathmap:${CMAKE_SOURCE_DIR}=src /pathmap:${CYDEC_SDK_SOURCE}=sdk)
endif()
option(CYDEC_NATIVE_DIAGNOSTICS "Private native exception diagnostics (development only)" OFF)
if(NOT CYDEC_SDK_SOURCE)
  message(FATAL_ERROR "Pass -DCYDEC_SDK_SOURCE=<verified CyMCPSDK sources>")
endif()
set(CYMCP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_LOCAL OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_CHAT_UI OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_CHAT_APP OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CYDEC_SDK_SOURCE}" sdk)
add_library(cydec_mcp SHARED bridge.cpp)
target_link_libraries(cydec_mcp PRIVATE CyMCP::Core CyMCP::Platform CyMCP::Http CyMCP::MCP CyMCP::AI)
target_compile_features(cydec_mcp PRIVATE cxx_std_20)
target_compile_definitions(cydec_mcp PRIVATE CYDEC_SDK_REVISION="${CYDEC_SDK_REVISION}")
if(MSVC)
  target_compile_options(cydec_mcp PRIVATE /utf-8 /Brepro /pathmap:${CMAKE_SOURCE_DIR}=src /pathmap:${CYDEC_SDK_SOURCE}=sdk)
  if(CYDEC_NATIVE_DIAGNOSTICS)
    target_compile_options(cydec_mcp PRIVATE /Zi)
    target_link_options(cydec_mcp PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF)
    target_link_libraries(cydec_mcp PRIVATE dbghelp)
    target_compile_definitions(cydec_mcp PRIVATE CYDEC_NATIVE_DIAGNOSTICS=1)
  endif()
endif()
