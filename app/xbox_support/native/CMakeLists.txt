cmake_minimum_required(VERSION 3.20)
project(CyXboxNativeTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT EXISTS "${XENONRECOMP_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set XENONRECOMP_SOURCE to the pinned XenonRecomp checkout")
endif()
add_subdirectory("${XENONRECOMP_SOURCE}" upstream EXCLUDE_FROM_ALL)
if(XBOX_NATIVE_PATCHED_RECOMPILER)
  get_target_property(recompiler_sources XenonRecomp SOURCES)
  set(replacement_sources "${XBOX_NATIVE_PATCHED_RECOMPILER}")
  foreach(source IN LISTS recompiler_sources)
    if(source STREQUAL "recompiler.cpp")
      continue()
    endif()
    if(IS_ABSOLUTE "${source}" OR source MATCHES "^\\$<")
      list(APPEND replacement_sources "${source}")
    else()
      list(APPEND replacement_sources "${XENONRECOMP_SOURCE}/XenonRecomp/${source}")
    endif()
  endforeach()
  set_property(TARGET XenonRecomp PROPERTY SOURCES "${replacement_sources}")
  target_include_directories(XenonRecomp PRIVATE "${XENONRECOMP_SOURCE}/XenonRecomp")
endif()
add_executable(xbox-image-probe image_probe.cpp)
set_property(TARGET xbox-image-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(xbox-image-probe PRIVATE NOMINMAX)
target_link_libraries(xbox-image-probe PRIVATE XenonUtils fmt::fmt)
